\begin{example}
$f(0)=0,f'(0)=f'(1)=0$,求证
\[ \z| \int_{0}^{1} f(x)\,dx \y| \le \frac{1}{3\sqrt{5}} 
\z( \int_{0}^{1} \z(f''(x)\y)^2 dx \y)^{\frac{1}{2}}   \]
\end{example}


\begin{proof}
\begin{align*}
\int_{0}^{1} f(x)\,dx &= -\int_{0}^{1} f(x) d(1-x) 
= \int_{0}^{1} f'(x)(1-x)\,dx - (1-x)f(x) \Big|_{0}^{1} \\
&= \int_{0}^{1} f'(x)(1-x)\,dx = \int_{0}^{1} f'(x) d\z( x -\frac{x^2}{2} +C\y)\\
&= \z( x -\frac{x^2}{2} +C\y) f'(x) \Big|_{0}^{1} - \int_{0}^{1} \z( x -\frac{x^2}{2} +C\y) f''(x)\,dx \\
&= \int_{0}^{1} \z(\frac{x^2}{2}-x+C\y) f''(x)\,dx 
\end{align*}
故
\[ \z| \int_{0}^{1} f(x)\,dx \y| \le \sqrt{\int_{0}^{1} \z(\frac{x^2}{2}-x+C\y)^2\,dx} \sqrt{ \int_{0}^{1} \z[f''(x)\y]^2 dx }   \]
下面确定系数C

\begin{align*}
\int_{0}^{1} \z(\frac{x^2}{2}-x+C\y)^2\,dx = &\int_{0}^{1}
  \z(\frac{x^2}{2}-x \y)^2 +2C\z(\frac{x^2}{2}-x \y) + C^2 \,dx \\
&=C^2 -\frac{2}{3}C +\frac{2}{15} \ge \frac{4ac-b^2}{4a} = \frac{\frac{8}{15}-\frac{9}{4}}{4} = \frac{1}{45}
\end{align*}

故取$C=-\frac{b}{2a}=\frac{1}{3} $时最佳系数为$\sqrt{\frac{1}{45}} = \frac{1}{3\sqrt{5}}$.
\end{proof}
